API CONVERSION REFERENCE
================================================================================

AUTHENTICATION:
  Old: import { signInWithPopup } from 'firebase/auth'
  New: firebase.auth.GoogleAuthProvider()

  Old: new GoogleAuthProvider()
  New: new firebase.auth.GoogleAuthProvider()

  Old: await signInWithPopup(auth, provider)
  New: auth.signInWithPopup(provider).then(...)

  Old: await signOut(auth)
  New: auth.signOut().then(...)

  Old: onAuthStateChanged(auth, callback)
  New: auth.onAuthStateChanged(callback)

FIRESTORE:
  Old: import { collection, addDoc } from 'firebase/firestore'
  New: db.collection(name)

  Old: await addDoc(collection(db, name), data)
  New: db.collection(name).add(data).then(...)

  Old: serverTimestamp()
  New: new Date()

  Old: enableIndexedDbPersistence(db)
  New: db.enablePersistence()

STORAGE:
  Old: import { ref, uploadBytes, getDownloadURL } from 'firebase/storage'
  New: storage.ref(path)

  Old: ref(storage, path)
  New: storage.ref(path)

  Old: await uploadBytes(ref, file)
  New: ref.put(file).then(...)

  Old: await getDownloadURL(ref)
  New: ref.getDownloadURL().then(...)
